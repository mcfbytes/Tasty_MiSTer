cmake_minimum_required(VERSION 3.21)
project(tasty VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(EXISTS "${CMAKE_SOURCE_DIR}/cmake/ccache.cmake")
  include(cmake/ccache.cmake)
endif()

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "" FORCE)
endif()

option(TASTY_TESTS "Build tasty host tests" ON)

include(cmake/decompressors.cmake)
find_path(TASTY_LZO_INCLUDE_DIR lzo/lzo1x.h)
find_library(TASTY_LZO_LIBRARY NAMES lzo2)
foreach(_dep IN ITEMS LIBCHDR MINIZIP LZO)
  if(NOT TASTY_${_dep}_INCLUDE_DIR OR NOT TASTY_${_dep}_LIBRARY)
    message(FATAL_ERROR "tasty needs ${_dep}: headers and library not both found")
  endif()
endforeach()

option(TASTY_WITH_HDOSD "Compile the Slint HD OSD raster" OFF)
set(TASTY_HDOSD_FONT_DIR "" CACHE PATH "Directory with NotoSans-Regular.ttf and NotoSans-Bold.ttf")
if(TASTY_WITH_HDOSD)
  if(NOT Slint_DIR AND TASTY_STATIC_DEPS AND EXISTS "${TASTY_STATIC_DEPS}/slint/lib/cmake/Slint")
    set(Slint_DIR "${TASTY_STATIC_DEPS}/slint/lib/cmake/Slint" CACHE PATH "Slint package" FORCE)
  endif()
  if(NOT TASTY_HDOSD_FONT_DIR AND TASTY_STATIC_DEPS)
    set(TASTY_HDOSD_FONT_DIR "${TASTY_STATIC_DEPS}/fonts" CACHE PATH "Noto Sans faces" FORCE)
  endif()

  if(NOT SLINT_COMPILER AND TASTY_STATIC_DEPS AND EXISTS "${TASTY_STATIC_DEPS}/slint")
    if(NOT EXISTS "${TASTY_STATIC_DEPS}/slint/bin/slint-compiler")
      message(FATAL_ERROR "${TASTY_STATIC_DEPS}/slint has no bin/slint-compiler: rerun build-static-slint.sh")
    endif()
    set(SLINT_COMPILER "${TASTY_STATIC_DEPS}/slint/bin/slint-compiler" CACHE STRING "slint-compiler" FORCE)
  endif()
  if(SLINT_COMPILER STREQUAL "download")
    message(FATAL_ERROR "SLINT_COMPILER=download fetches an unpinned binary: name a slint-compiler")
  endif()
  find_package(Slint 1.18 REQUIRED)
  if(NOT EXISTS "${TASTY_HDOSD_FONT_DIR}/NotoSans-Regular.ttf"
     OR NOT EXISTS "${TASTY_HDOSD_FONT_DIR}/NotoSans-Bold.ttf")
    message(FATAL_ERROR "the HD OSD needs NotoSans-Regular.ttf and NotoSans-Bold.ttf in TASTY_HDOSD_FONT_DIR "
                        "(src/tools/build-static-deps.py fetches them)")
  endif()
endif()
message(STATUS "HD OSD: ${TASTY_WITH_HDOSD}")

include(cmake/warnings.cmake)
include(cmake/header_checks.cmake)
include(cmake/unit_test.cmake)
enable_testing()

if(NOT TASTY_VDATE)
  string(TIMESTAMP TASTY_VDATE_DEFAULT "%y%m%d" UTC)
  set(TASTY_VDATE "${TASTY_VDATE_DEFAULT}" CACHE STRING "YYMMDD version stamp")
endif()

set(TASTY_SPLASH_PNG "${CMAKE_SOURCE_DIR}/art/banner.png" CACHE FILEPATH "splash PNG")

add_subdirectory(src/infra)
add_subdirectory(src/l0_os)
add_subdirectory(src/l0_hal)
add_subdirectory(src/l0_reactor)
add_subdirectory(src/l0_boards)
add_subdirectory(src/l1_proto)
add_subdirectory(src/l2_services)
add_subdirectory(src/l3_cores)
add_subdirectory(src/l4_app)
add_subdirectory(src/l5_ui)
add_subdirectory(src/firmware)

if(TASTY_TESTS)
  add_subdirectory(tests)
endif()
